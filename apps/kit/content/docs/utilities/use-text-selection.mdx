## Usage

```tsx
import { useTextSelection } from "@/hooks/use-text-selection"
```

```tsx
const read = useTextSelection({ containerRef, paragraphs, wordSelect })
const selection = read()

if (selection) {
  const { para, start, end } = selection
}
```
