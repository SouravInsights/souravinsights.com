## Minimal article

The engine takes a container of paragraphs and reports marks as plain anchors.
Keep the marks in React state until you have somewhere to store them.

```tsx
"use client"

import { useRef, useState } from "react"
import { Highlight, type HighlightMark } from "@/components/highlight/highlight"
import { ShareQuoteDialog } from "@/components/quote/dialog"

function Article({ title, publication }: { title: string; publication: string }) {
  const ref = useRef<HTMLDivElement>(null)
  const [marks, setMarks] = useState<HighlightMark[]>([])
  const [quote, setQuote] = useState<{ quote: string; context: string } | null>(null)

  return (
    <>
      <div ref={ref}>
        <p>The best interfaces disappear.</p>
      </div>

      <Highlight
        containerRef={ref}
        editable
        marks={marks}
        onMarksChange={setMarks}
        onShareQuote={setQuote}
      />

      <ShareQuoteDialog
        open={quote !== null}
        onOpenChange={(open) => !open && setQuote(null)}
        quote={quote?.quote ?? ""}
        context={quote?.context ?? ""}
        title={title}
        publication={publication}
      />
    </>
  )
}
```

## Custom toolbar

`Highlight` renders the default toolbar unless you pass children. Every part is
exported, so you can keep the palette and the list and drop the rest.

```tsx
<Highlight
  containerRef={ref}
  editable
  marks={marks}
  onMarksChange={setMarks}
  onShareQuote={setQuote}
>
  <HighlightToolbar>
    <HighlightColors />
    <HighlightList />
  </HighlightToolbar>
</Highlight>
```

## Persisted marks

Pass `initialMarks` to restore stored anchors, and the callbacks to write them
back. `onCreate` returns the row id, so a recolor or delete can target it.

```tsx
<Highlight
  containerRef={ref}
  editable
  initialMarks={saved}
  onCreate={async (anchor) => api.createHighlight(anchor)}
  onRecolor={(id, hue) => api.recolorHighlight(id, hue)}
  onDelete={(id) => api.deleteHighlight(id)}
  onShareQuote={setQuote}
/>
```

## The whole thing

`article-highlights` is the block that wires the engine, the rail and the
studio together. Start there, then peel pieces off as you need to.

```tsx
import { ArticleHighlights } from "@/components/blocks/article-highlights"

<ArticleHighlights title={title} publication={publication}>
  {content}
</ArticleHighlights>
```
